Recommended Fix Pattern

ℹ Option A is the recommended approach. Add per-env tfvars overrides for components that already have a retention variable. No code changes required for the highest-priority fix (cloudwatch_loggroup).
terraform/components/cloudwatch_loggroup/tfvars/dev-ex.tfvars — new file
# DEV baseline: 90 days (3 months)
retention_in_days = 90
terraform/components/cloudwatch_loggroup/tfvars/stg-ex.tfvars — new file
# STG baseline: 180 days (6 months)
retention_in_days = 180
terraform/components/elasticache/tfvars/dev-ex.tfvars — add line
# Add to existing file (node_type already present)
log_retention_days = 90
terraform/components/elasticache/tfvars/stg-ex.tfvars — add line
log_retention_days = 180
terraform/components/waf-maintenance-lambda/tfvars/dev-ex.tfvars — add line
log_retention_in_days = 90
terraform/components/waf-maintenance-lambda/tfvars/stg-ex.tfvars — add line
log_retention_in_days = 180
terraform/components/vpc/log_to_cloudwatch.tf — extract hardcoded value (code change)
# Before (hardcoded)
resource "aws_cloudwatch_log_group" "flow_log" {
  name              = "/aws/vpc-flow-log/${var.vpc_name}"
  retention_in_days = 30  # ← hardcoded
}

# After (variable-driven)
resource "aws_cloudwatch_log_group" "flow_log" {
  name              = "/aws/vpc-flow-log/${var.vpc_name}"
  retention_in_days = var.log_retention_days
}

# Add to variables.tf
variable "log_retention_days" {
  type        = number
  description = "CloudWatch log group retention in days. Baseline: DEV=90, STG=180, PRD=365."
  validation {
    condition     = contains([7, 14, 30, 60, 90, 120, 150, 180, 365, 731, 0], var.log_retention_days)
    error_message = "Must be a valid CloudWatch Logs retention period value."
  }
}
terraform/components/cloudwatch_circuitbreaker_lambda/log_group.tf — new file
# Add managed log group before Lambda to ensure retention is set
resource "aws_cloudwatch_log_group" "this" {
  name              = "/aws/lambda/${aws_lambda_function.circuitbreaker_to_slack.function_name}"
  retention_in_days = var.log_retention_days
}

# In main.tf — add depends_on to the Lambda resource
resource "aws_lambda_function" "circuitbreaker_to_slack" {
  # ... existing config ...
  depends_on = [aws_cloudwatch_log_group.this]
}
terraform/components/cloudwatch_errorlog_lambda/log_group.tf — new file
# Same pattern for error-log-to-slack Lambda
resource "aws_cloudwatch_log_group" "this" {
  name              = "/aws/lambda/${aws_lambda_function.error_log_to_slack.function_name}"
  retention_in_days = var.log_retention_days
}

resource "aws_lambda_function" "error_log_to_slack" {
  # ... existing config ...
  depends_on = [aws_cloudwatch_log_group.this]
}

# variables.tf — standardized var name (matches newer components)
variable "log_retention_days" {
  type        = number
  description = "CloudWatch log group retention in days. Baseline: DEV=90, STG=180, PRD=365."
  validation {
    condition     = contains([7, 14, 30, 60, 90, 120, 150, 180, 365, 731, 0], var.log_retention_days)
    error_message = "Must be a valid CloudWatch Logs retention period value."
  }
}
